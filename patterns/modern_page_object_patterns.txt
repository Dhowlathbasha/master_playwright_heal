  readonly passwordInput: Locator;
  readonly loginButton: Locator;

  readonly rememberMeCheckbox: Locator;
  readonly forgotPasswordLink: Locator;

    await this.loginButton.click();

    return await this.errorMessage.textContent();
  }

    await this.forgotPasswordLink.click();

    await this.page.waitForURL('**/dashboard');
  }

  async expectToBeVisible() {
    await expect(this.usernameInput).toBeVisible();
    await expect(this.passwordInput).toBeVisible();
    await expect(this.loginButton).toBeVisible();
  }
}

    this.rowsPerPageSelect = page.getByLabel('Rows per page');
    this.loadingSpinner = page.getByTestId('loading-spinner');
  }

  async goto() {
    await this.page.goto('/data-table');
  }

  async searchTable(searchTerm: string) {
    await this.searchInput.fill(searchTerm);
    await this.waitForTableUpdate();
  }

  async sortByColumn(columnName: string) {
    await this.page.getByRole('button', { name: `Sort by ${columnName}` }).click();
    await this.waitForTableUpdate();
  }

  async goToNextPage() {
    await this.paginationNext.click();
    await this.waitForTableUpdate();
  }

  async selectRowsPerPage(count: number) {
    await this.rowsPerPageSelect.selectOption(count.toString());
    await this.waitForTableUpdate();
  }

  async getRowData(rowIndex: number) {
    const row = this.table.locator('tbody tr').nth(rowIndex);
    const cells = row.locator('td');
    const cellCount = await cells.count();
    const data: string[] = [];
    
    for (let i = 0; i < cellCount; i++) {
      const cellText = await cells.nth(i).textContent();
      data.push(cellText || '');
    }
    
    return data;
  }

  async clickRowAction(rowIndex: number, action: string) {
    const row = this.table.locator('tbody tr').nth(rowIndex);
    await row.getByRole('button', { name: action }).click();
  }

  async waitForTableUpdate() {
    await this.loadingSpinner.waitFor({ state: 'visible' });
    await this.loadingSpinner.waitFor({ state: 'hidden' });
  }

  async expectTableToContainData() {
    await expect(this.table.locator('tbody tr')).toHaveCount({ min: 1 });
  }
}

    await this.page.goto('/upload');
  }

    await this.fileInput.setInputFiles(filePath);
    await this.uploadButton.click();
  }

  async uploadMultipleFiles(filePaths: string[]) {
    await this.fileInput.setInputFiles(filePaths);
    await this.uploadButton.click();
  }

  async createTestFile(fileName: string, content: string = 'Test file content') {
    const testFilePath = path.join(__dirname, '..', 'test-data', fileName);
    await fs.writeFile(testFilePath, content);
    return testFilePath;
  }

  async dragAndDropFile(filePath: string) {
    const buffer = await fs.readFile(filePath);
    
    await this.dropZone.evaluate((element, { fileName, fileContent }) => {
      const dt = new DataTransfer();
      const file = new File([fileContent], fileName, { type: 'text/plain' });
      dt.items.add(file);
      
      const event = new DragEvent('drop', {
        bubbles: true,
        cancelable: true,
        dataTransfer: dt
      });
      
      element.dispatchEvent(event);
    }, { fileName: path.basename(filePath), fileContent: buffer.toString() });
  }

  async expectUploadProgress() {
    await expect(this.progressBar).toBeVisible();
    await expect(this.progressBar).toHaveAttribute('aria-valuenow');
  }

  async expectUploadSuccess() {
    await expect(this.uploadStatus).toHaveText('Upload successful');
    await expect(this.progressBar).not.toBeVisible();
  }

  async expectUploadError(errorMessage?: string) {
    await expect(this.uploadStatus).toContainText('Upload failed');
    if (errorMessage) {
      await expect(this.uploadStatus).toContainText(errorMessage);
    }
  }

  async getUploadedFiles() {
    return await this.fileList.locator('li').allTextContents();
  }

  async removeFile(fileName: string) {
    const fileItem = this.fileList.getByText(fileName).locator('..');
    await fileItem.getByRole('button', { name: 'Remove' }).click();
  }
}

// pages/PerformanceMonitoringPage.ts
import { Page, expect } from '@playwright/test';

export class PerformanceMonitoringPage {
  readonly page: Page;

  constructor(page: Page) {
    this.page = page;
  }

  async goto() {
    await this.page.goto('/performance');
  }

  async measurePageLoadTime() {
    const startTime = Date.now();
    await this.page.waitForLoadState('networkidle');
    const loadTime = Date.now() - startTime;
    return loadTime;
  }

  async getCoreWebVitals() {
    return await this.page.evaluate(() => {
      return new Promise((resolve) => {
        const vitals: any = {};
        
        new PerformanceObserver((list) => {
          for (const entry of list.getEntries()) {
            if (entry.name === 'first-contentful-paint') {
              vitals.fcp = entry.startTime;
            }
            if (entry.name === 'largest-contentful-paint') {
              vitals.lcp = entry.startTime;
            }
          }
        }).observe({ entryTypes: ['paint', 'largest-contentful-paint'] });

        // Get Cumulative Layout Shift
        new PerformanceObserver((list) => {
          let clsValue = 0;
          for (const entry of list.getEntries()) {
            if (!entry.hadRecentInput) {
              clsValue += (entry as any).value;
            }
          }
          vitals.cls = clsValue;
        }).observe({ entryTypes: ['layout-shift'] });

        setTimeout(() => resolve(vitals), 2000);
      });
    });
  }

  async getNetworkMetrics() {
    return await this.page.evaluate(() => {
      const navigation = performance.getEntriesByType('navigation')[0] as PerformanceNavigationTiming;
      return {
        domContentLoaded: navigation.domContentLoadedEventEnd - navigation.domContentLoadedEventStart,
        loadComplete: navigation.loadEventEnd - navigation.loadEventStart,
        firstByte: navigation.responseStart - navigation.requestStart,
        domInteractive: navigation.domInteractive - navigation.navigationStart
      };
    });
  }

  async getResourceMetrics() {
    return await this.page.evaluate(() => {
      const resources = performance.getEntriesByType('resource');
      return resources.map(resource => ({
        name: resource.name,
        duration: resource.duration,
        transferSize: (resource as any).transferSize,
        type: (resource as any).initiatorType
      }));
    });
  }

  async expectPageLoadUnder(maxTime: number) {
    const loadTime = await this.measurePageLoadTime();
    expect(loadTime).toBeLessThan(maxTime);
  }

  async expectGoodCoreWebVitals() {
    const vitals = await this.getCoreWebVitals();
    
    // FCP should be under 1.8s
    expect(vitals.fcp).toBeLessThan(1800);
    
    // LCP should be under 2.5s  
    expect(vitals.lcp).toBeLessThan(2500);
    
    // CLS should be under 0.1
    expect(vitals.cls).toBeLessThan(0.1);
  }

  async expectFastNetworkMetrics() {
    const metrics = await this.getNetworkMetrics();
    
    expect(metrics.firstByte).toBeLessThan(600); // TTFB under 600ms
    expect(metrics.domContentLoaded).toBeLessThan(1000); // DCL under 1s
  }
}
