// api/BaseApiClient.ts
import { APIRequestContext, expect } from '@playwright/test';

export class BaseApiClient {
  protected apiContext: APIRequestContext;
  protected baseURL: string;

  constructor(apiContext: APIRequestContext, baseURL: string = '') {
    this.apiContext = apiContext;
    this.baseURL = baseURL;
  }

  async get(endpoint: string, options: any = {}) {
    const response = await this.apiContext.get(`${this.baseURL}${endpoint}`, options);
    return response;
  }

  async post(endpoint: string, data: any = {}, options: any = {}) {
    const response = await this.apiContext.post(`${this.baseURL}${endpoint}`, {
      data,
      ...options
    });
    return response;
  }

  async put(endpoint: string, data: any = {}, options: any = {}) {
    const response = await this.apiContext.put(`${this.baseURL}${endpoint}`, {
      data,
      ...options
    });
    return response;
  }

  async patch(endpoint: string, data: any = {}, options: any = {}) {
    const response = await this.apiContext.patch(`${this.baseURL}${endpoint}`, {
      data,
      ...options
    });
    return response;
  }

  async delete(endpoint: string, options: any = {}) {
    const response = await this.apiContext.delete(`${this.baseURL}${endpoint}`, options);
    return response;
  }

  async expectSuccessResponse(response: any, expectedStatus: number = 200) {
    expect(response.status()).toBe(expectedStatus);
    expect(response.ok()).toBeTruthy();
  }

  async expectErrorResponse(response: any, expectedStatus: number) {
    expect(response.status()).toBe(expectedStatus);
    expect(response.ok()).toBeFalsy();
  }

  async getResponseBody(response: any) {
    return await response.json();
  }

  async validateResponseSchema(response: any, schema: any) {
    const body = await this.getResponseBody(response);
    // Add JSON schema validation here if needed
    return body;
  }
}

// api/UserApiClient.ts
import { APIRequestContext } from '@playwright/test';
import { BaseApiClient } from './BaseApiClient';

export interface User {

  name: string;
  email: string;
  role?: string;
  active?: boolean;

  updatedAt?: string;

  users: User[];
  pagination: {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
  };

  constructor(apiContext: APIRequestContext) {
    super(apiContext, '/api');
  }

  async createUser(userData: Omit<User, 'id' | 'createdAt' | 'updatedAt'>) {
    const response = await this.post('/users', userData);
    await this.expectSuccessResponse(response, 201);
    return await this.getResponseBody(response) as User;
  }

  async getUserById(userId: string) {
    const response = await this.get(`/users/${userId}`);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async getAllUsers(params: {
    page?: number;
    limit?: number;
    role?: string;
    active?: boolean;
    search?: string;

    const queryParams = new URLSearchParams();
    Object.entries(params).forEach(([key, value]) => {
      if (value !== undefined) {
        queryParams.append(key, value.toString());
      }
    });

    const endpoint = `/users${queryParams.toString() ? `?${queryParams}` : ''}`;
    const response = await this.get(endpoint);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as UserListResponse;
  }

  async updateUser(userId: string, userData: Partial<User>) {
    const response = await this.put(`/users/${userId}`, userData);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async patchUser(userId: string, userData: Partial<User>) {
    const response = await this.patch(`/users/${userId}`, userData);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async deleteUser(userId: string) {
    const response = await this.delete(`/users/${userId}`);
    await this.expectSuccessResponse(response, 204);
  }

  async activateUser(userId: string) {
    const response = await this.post(`/users/${userId}/activate`);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async deactivateUser(userId: string) {
    const response = await this.post(`/users/${userId}/deactivate`);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async validateUserExists(email: string) {
    const users = await this.getAllUsers({ search: email });
    return users.users.some(user => user.email === email);
  }

  async expectUserNotFound(userId: string) {
    const response = await this.get(`/users/${userId}`);
    await this.expectErrorResponse(response, 404);
  }
}

// api/AuthApiClient.ts
import { APIRequestContext } from '@playwright/test';
import { BaseApiClient } from './BaseApiClient';

export interface LoginCredentials {
  email: string;
  password: string;
  rememberMe?: boolean;
}

export interface AuthResponse {
  token: string;
  refreshToken: string;
  user: {
    id: string;
    name: string;
    email: string;
    role: string;
  };
  expiresIn: number;
}

export interface RegisterData {
  name: string;
  email: string;
  password: string;
  confirmPassword: string;
}

export class AuthApiClient extends BaseApiClient {
  private authToken: string | null = null;

  constructor(apiContext: APIRequestContext) {
    super(apiContext, '/api/auth');
  }

  async login(credentials: LoginCredentials) {
    const response = await this.post('/login', credentials);
    await this.expectSuccessResponse(response);
    
    const authData = await this.getResponseBody(response) as AuthResponse;
    this.authToken = authData.token;
    
    return authData;
  }

  async register(userData: RegisterData) {
    const response = await this.post('/register', userData);
    await this.expectSuccessResponse(response, 201);
    return await this.getResponseBody(response) as AuthResponse;
  }

  async logout() {
    const response = await this.post('/logout', {}, {
      headers: this.getAuthHeaders()
    });
    await this.expectSuccessResponse(response);
    this.authToken = null;
  }

  async refreshToken(refreshToken: string) {
    const response = await this.post('/refresh', { refreshToken });
    await this.expectSuccessResponse(response);
    
    const authData = await this.getResponseBody(response) as AuthResponse;
    this.authToken = authData.token;
    
    return authData;
  }

  async getCurrentUser() {
    const response = await this.get('/me', {
      headers: this.getAuthHeaders()
    });
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response);
  }

  async changePassword(currentPassword: string, newPassword: string) {
    const response = await this.post('/change-password', {
      currentPassword,
      newPassword
    }, {
      headers: this.getAuthHeaders()
    });
    await this.expectSuccessResponse(response);
  }

  async requestPasswordReset(email: string) {
    const response = await this.post('/password-reset-request', { email });
    await this.expectSuccessResponse(response);
  }

  async resetPassword(token: string, newPassword: string) {
    const response = await this.post('/password-reset', {
      token,
      newPassword
    });
    await this.expectSuccessResponse(response);
  }

  async verifyEmail(token: string) {
    const response = await this.post('/verify-email', { token });
    await this.expectSuccessResponse(response);
  }

  getAuthToken() {
    return this.authToken;
  }

  private getAuthHeaders() {
    return this.authToken ? { 'Authorization': `Bearer ${this.authToken}` } : {};
  }

  async expectUnauthorized(endpoint: string) {
    const response = await this.get(endpoint);
    await this.expectErrorResponse(response, 401);
  }

  async expectForbidden(endpoint: string) {
    const response = await this.get(endpoint, {
      headers: this.getAuthHeaders()
    });
    await this.expectErrorResponse(response, 403);
  }
}

// api/MicroservicesApiClient.ts
import { APIRequestContext } from '@playwright/test';
import { BaseApiClient } from './BaseApiClient';

export interface ServiceHealth {
  service: string;
  status: 'healthy' | 'unhealthy' | 'degraded';
  timestamp: string;
  dependencies: string[];
}

export class MicroservicesApiClient extends BaseApiClient {
  private services: Map<string, string> = new Map();

  constructor(apiContext: APIRequestContext) {
    super(apiContext);
    this.initializeServices();
  }

  private initializeServices() {
    this.services.set('user-service', '/api/users');
    this.services.set('product-service', '/api/products');
    this.services.set('order-service', '/api/orders');
    this.services.set('payment-service', '/api/payments');
    this.services.set('notification-service', '/api/notifications');
  }

  // Health check for all services
  async checkAllServicesHealth() {
    const healthChecks = [];

    for (const [serviceName, basePath] of this.services) {
      healthChecks.push(this.checkServiceHealth(serviceName, basePath));
    }

    return await Promise.allSettled(healthChecks);
  }

  async checkServiceHealth(serviceName: string, basePath: string) {
    try {
      const response = await this.get(`${basePath}/health`);
      
      if (response.ok()) {
        const healthData = await this.getResponseBody(response);
        return {
          service: serviceName,
          status: 'healthy',
          data: healthData
        };
      } else {
        return {
          service: serviceName,
          status: 'unhealthy',
          statusCode: response.status()
        };
      }
    } catch (error) {
      return {
        service: serviceName,
        status: 'error',
        error: error.message
      };
    }
  }

  // Cross-service integration testing

    const registeredServices = services.map((s: any) => s.name);

  // Circuit breaker testing
  async testCircuitBreaker(serviceName: string, faultInjectionCount: number = 5) {
    const basePath = this.services.get(serviceName);
    if (!basePath) {
      throw new Error(`Service ${serviceName} not found`);
    }

    const results = [];

    // Inject faults to trigger circuit breaker
    for (let i = 0; i < faultInjectionCount; i++) {
      const response = await this.get(`${basePath}/fault-injection`);
      results.push({
        attempt: i + 1,
        status: response.status(),
        circuitOpen: response.headers()['x-circuit-breaker'] === 'open'
      });
    }

    // Test if circuit breaker is now open
    const testResponse = await this.get(`${basePath}/test`);
    results.push({
      attempt: 'final-test',
      status: testResponse.status(),
      circuitOpen: testResponse.status() === 503
    });

    return results;
  }

  // Load balancing testing
  async testLoadBalancing(serviceName: string, requestCount: number = 20) {
    const basePath = this.services.get(serviceName);
    if (!basePath) {
      throw new Error(`Service ${serviceName} not found`);
    }

    const promises = [];
    for (let i = 0; i < requestCount; i++) {
      promises.push(this.get(`${basePath}/instance-info`));
    }

    const responses = await Promise.all(promises);
    const instanceCounts = new Map();

    for (const response of responses) {
      if (response.ok()) {
        const data = await response.json();
        const instanceId = data.instanceId;
        instanceCounts.set(instanceId, (instanceCounts.get(instanceId) || 0) + 1);
      }
    }

    return {
      totalRequests: requestCount,
      instanceDistribution: Object.fromEntries(instanceCounts),
      isLoadBalanced: instanceCounts.size > 1
    };
  }

// api/SecurityApiClient.ts
import { APIRequestContext } from '@playwright/test';
import { BaseApiClient } from './BaseApiClient';

export interface SecurityTestResult {
  testName: string;
  vulnerable: boolean;
  details: any;
  severity: 'low' | 'medium' | 'high' | 'critical';
}

export class SecurityApiClient extends BaseApiClient {
  constructor(apiContext: APIRequestContext) {
    super(apiContext, '/api');
  }

  // OWASP API Security Top 10 testing
  async runOWASPTop10Tests(endpoint: string) {
    const tests = [
      this.testBrokenObjectLevelAuthorization(endpoint),
      this.testBrokenUserAuthentication(endpoint),
      this.testExcessiveDataExposure(endpoint),
      this.testLackOfResourcesRateLimiting(endpoint),
      this.testBrokenFunctionLevelAuthorization(endpoint),
      this.testMassAssignment(endpoint),
      this.testSecurityMisconfiguration(endpoint),
      this.testInjection(endpoint),
      this.testImproperAssetsManagement(endpoint),
      this.testInsufficientLoggingMonitoring(endpoint)
    ];

    const results = await Promise.allSettled(tests);
    return results.map((result, index) => ({
      testIndex: index + 1,
      status: result.status,
      data: result.status === 'fulfilled' ? result.value : result.reason
    }));
  }

  // API1:2023 Broken Object Level Authorization
  async testBrokenObjectLevelAuthorization(endpoint: string): Promise<SecurityTestResult> {
    const testCases = [
      { userId: '1', expectedAccess: true },
      { userId: '999', expectedAccess: false }, // Non-existent user
      { userId: '../admin', expectedAccess: false }, // Path traversal
      { userId: '2', expectedAccess: false } // Different user's data
    ];

    const vulnerabilities = [];

    for (const testCase of testCases) {
      const response = await this.get(`${endpoint}/${testCase.userId}`, {
        headers: { 'Authorization': 'Bearer user-1-token' }
      });

      if (response.ok() && !testCase.expectedAccess) {
        vulnerabilities.push(`Unauthorized access to user ${testCase.userId}`);
      }
    }

    return {
      testName: 'Broken Object Level Authorization',
      vulnerable: vulnerabilities.length > 0,
      details: vulnerabilities,
      severity: vulnerabilities.length > 0 ? 'high' : 'low'
    };
  }

  // API2:2023 Broken Authentication
  async testBrokenUserAuthentication(endpoint: string): Promise<SecurityTestResult> {
    const weakPasswords = ['123456', 'password', 'admin', ''];
    const vulnerabilities = [];

    for (const password of weakPasswords) {
      const response = await this.post(`${endpoint}/login`, {
        username: 'admin',
        password: password
      });

      if (response.ok()) {
        vulnerabilities.push(`Weak password accepted: "${password}"`);
      }
    }

    // Test for brute force protection
    const bruteForceAttempts = [];
    for (let i = 0; i < 10; i++) {
      const response = await this.post(`${endpoint}/login`, {
        username: 'admin',
        password: 'wrong-password'
      });
      bruteForceAttempts.push(response.status());
    }

    if (bruteForceAttempts.every(status => status !== 429)) {
      vulnerabilities.push('No rate limiting on authentication endpoint');
    }

    return {
      testName: 'Broken Authentication',
      vulnerable: vulnerabilities.length > 0,
      details: vulnerabilities,
      severity: vulnerabilities.length > 0 ? 'critical' : 'low'
    };
  }

  // API3:2023 Excessive Data Exposure
  async testExcessiveDataExposure(endpoint: string): Promise<SecurityTestResult> {
    const response = await this.get(`${endpoint}/users/profile`);
    
    if (response.ok()) {
      const data = await response.json();
      const sensitiveFields = ['password', 'ssn', 'creditCard', 'bankAccount', 'privateKey'];
      const exposedSensitiveData = [];

      const checkObject = (obj: any, path = '') => {
        Object.keys(obj).forEach(key => {
          const currentPath = path ? `${path}.${key}` : key;
          
          if (sensitiveFields.some(field => key.toLowerCase().includes(field.toLowerCase()))) {
            exposedSensitiveData.push(currentPath);
          }
          
          if (typeof obj[key] === 'object' && obj[key] !== null) {
            checkObject(obj[key], currentPath);
          }
        });
      };

      checkObject(data);

      return {
        testName: 'Excessive Data Exposure',
        vulnerable: exposedSensitiveData.length > 0,
        details: exposedSensitiveData,
        severity: exposedSensitiveData.length > 0 ? 'medium' : 'low'
      };
    }

    return {
      testName: 'Excessive Data Exposure',
      vulnerable: false,
      details: 'Could not access endpoint',
      severity: 'low'
    };
  }

  // API4:2023 Lack of Resources & Rate Limiting
  async testLackOfResourcesRateLimiting(endpoint: string): Promise<SecurityTestResult> {
    const rapidRequests = [];
    const startTime = Date.now();

    // Send 100 requests rapidly
    for (let i = 0; i < 100; i++) {
      rapidRequests.push(this.get(endpoint));
    }

    const responses = await Promise.allSettled(rapidRequests);
    const endTime = Date.now();

    const successfulRequests = responses.filter(
      r => r.status === 'fulfilled' && r.value.ok()
    ).length;

    const rateLimitedRequests = responses.filter(
      r => r.status === 'fulfilled' && r.value.status() === 429

      testName: 'Lack of Resources & Rate Limiting',
      vulnerable,
      details: {
        totalRequests: 100,
        successfulRequests,
        rateLimitedRequests,
        duration: endTime - startTime
      },
      severity: vulnerable ? 'high' : 'low'
    };
  }
