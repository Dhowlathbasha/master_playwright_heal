// step-definitions/parameterized.steps.ts
import { Given, When, Then, DataTable } from '@cucumber/cucumber';
import { test, expect } from '../fixtures/test-fixtures';

// Parameterized scenarios support
When('I perform the following actions:', async ({ page }, dataTable: DataTable) => {
  const actions = dataTable.hashes();
  
  for (const action of actions) {
    switch (action.action.toLowerCase()) {
      case 'click':
        await page.getByRole('button', { name: action.target }).click();
        break;
      case 'fill':
        await page.getByLabel(action.target).fill(action.value);
        break;
      case 'select':
        await page.getByLabel(action.target).selectOption(action.value);
        break;
      case 'check':
        await page.getByLabel(action.target).check();
        break;
      case 'uncheck':
        await page.getByLabel(action.target).uncheck();
        break;
    }
  }
});

When('I verify the following elements:', async ({ page }, dataTable: DataTable) => {
  const verifications = dataTable.hashes();
  
  for (const verification of verifications) {
    const element = page.getByTestId(verification.element);
    
    switch (verification.check.toLowerCase()) {
      case 'visible':
        await expect(element).toBeVisible();
        break;
      case 'hidden':
        await expect(element).not.toBeVisible();
        break;
      case 'enabled':
        await expect(element).toBeEnabled();
        break;
      case 'disabled':
        await expect(element).toBeDisabled();
        break;
      case 'contains':
        await expect(element).toContainText(verification.value);
        break;
    }
  }
});

Then('the page should meet the following criteria:', async ({ page }, dataTable: DataTable) => {
  const criteria = dataTable.rowsHash();
  
  if (criteria['Title']) {
    await expect(page).toHaveTitle(new RegExp(criteria['Title']));

  }
