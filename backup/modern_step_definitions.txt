import { Given, When, Then, DataTable } from '@cucumber/cucumber';
import { test, expect } from '../fixtures/test-fixtures';
import { LoginPage } from '../pages/LoginPage';

let loginPage: LoginPage;

test.beforeEach(async ({ loginPage: lp }) => {
  loginPage = lp;
});

// Data table examples
When('I test login with multiple credentials:', async (dataTable: DataTable) => {
  const credentials = dataTable.hashes();
  
  for (const credential of credentials) {
    await loginPage.goto();
    await loginPage.login(credential.username, credential.password);
    
    if (credential.expected === 'success') {
      await loginPage.waitForLoginSuccess();
    } else {
      const errorMessage = await loginPage.getErrorMessage();
      expect(errorMessage).toBeTruthy();
    }
  }
});

// Scenario outline support
When('I login with username {string} and password {string}', async (username: string, password: string) => {
  await loginPage.login(username, password);
});

Then('the login result should be {string}', async (expectedResult: string) => {
  if (expectedResult === 'success') {
    await loginPage.waitForLoginSuccess();
  } else {
    const errorMessage = await loginPage.getErrorMessage();
    expect(errorMessage).toContain('Invalid credentials');
  }
});

import { Given, When, Then } from '@cucumber/cucumber';
import { test, expect } from '../fixtures/test-fixtures';
import { request, APIRequestContext } from '@playwright/test';

let apiContext: APIRequestContext;

test.beforeAll(async () => {
  apiContext = await request.newContext({
    baseURL: 'https://api.example.com',
    extraHTTPHeaders: {
      'Authorization': 'Bearer your-api-token'
    }
  });
});

test.afterAll(async () => {
  await apiContext.dispose();
});

// API setup steps
Given('the API returns user data for user {string}', async (userId: string) => {
  const response = await apiContext.get(`/users/${userId}`);
  expect(response.ok()).toBeTruthy();
});

Given('I create a test user via API', async () => {
  const response = await apiContext.post('/users', {
    data: {
      name: 'Test User',
      email: 'test@example.com'
    }
  });
  expect(response.ok()).toBeTruthy();
});

// API validation steps
Then('the API should confirm the data was updated', async () => {
  const response = await apiContext.get('/users/profile');
  const userData = await response.json();
  expect(userData.lastModified).toBeTruthy();
});

When('I verify the cart contents via API', async () => {
  const response = await apiContext.get('/cart');

  expect(cartData.items.length).toBeGreaterThan(0);

    localStorage.setItem('cart', JSON.stringify(['item1', 'item2']));
  });

  await page.context().clearCookies();
  await page.evaluate(() => {
    localStorage.clear();
    sessionStorage.clear();
  });

  const cartItems = await page.evaluate(() => localStorage.getItem('cart'));
