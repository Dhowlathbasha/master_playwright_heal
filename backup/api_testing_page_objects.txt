// api/BaseApiClient.ts
import { APIRequestContext, expect } from '@playwright/test';

export class BaseApiClient {
  protected apiContext: APIRequestContext;
  protected baseURL: string;

  constructor(apiContext: APIRequestContext, baseURL: string = '') {
    this.apiContext = apiContext;
    this.baseURL = baseURL;
  }

  async get(endpoint: string, options: any = {}) {
    const response = await this.apiContext.get(`${this.baseURL}${endpoint}`, options);
    return response;
  }

  async post(endpoint: string, data: any = {}, options: any = {}) {
    const response = await this.apiContext.post(`${this.baseURL}${endpoint}`, {
      data,
      ...options
    });
    return response;
  }

  async put(endpoint: string, data: any = {}, options: any = {}) {
    const response = await this.apiContext.put(`${this.baseURL}${endpoint}`, {
      data,
      ...options
    });
    return response;
  }

  async patch(endpoint: string, data: any = {}, options: any = {}) {
    const response = await this.apiContext.patch(`${this.baseURL}${endpoint}`, {
      data,
      ...options
    });
    return response;
  }

  async delete(endpoint: string, options: any = {}) {
    const response = await this.apiContext.delete(`${this.baseURL}${endpoint}`, options);
    return response;
  }

  async expectSuccessResponse(response: any, expectedStatus: number = 200) {
    expect(response.status()).toBe(expectedStatus);
    expect(response.ok()).toBeTruthy();
  }

  async expectErrorResponse(response: any, expectedStatus: number) {
    expect(response.status()).toBe(expectedStatus);
    expect(response.ok()).toBeFalsy();
  }

  async getResponseBody(response: any) {
    return await response.json();
  }

  async validateResponseSchema(response: any, schema: any) {
    const body = await this.getResponseBody(response);
    // Add JSON schema validation here if needed
    return body;
  }
}

// api/UserApiClient.ts
import { APIRequestContext } from '@playwright/test';
import { BaseApiClient } from './BaseApiClient';

export interface User {
  id?: string;
  name: string;
  email: string;
  role?: string;
  active?: boolean;
  createdAt?: string;
  updatedAt?: string;
}

export interface UserListResponse {
  users: User[];
  pagination: {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
  };
}

export class UserApiClient extends BaseApiClient {
  constructor(apiContext: APIRequestContext) {
    super(apiContext, '/api');
  }

  async createUser(userData: Omit<User, 'id' | 'createdAt' | 'updatedAt'>) {
    const response = await this.post('/users', userData);
    await this.expectSuccessResponse(response, 201);
    return await this.getResponseBody(response) as User;
  }

  async getUserById(userId: string) {
    const response = await this.get(`/users/${userId}`);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async getAllUsers(params: {
    page?: number;
    limit?: number;
    role?: string;
    active?: boolean;
    search?: string;
  } = {}) {
    const queryParams = new URLSearchParams();
    Object.entries(params).forEach(([key, value]) => {
      if (value !== undefined) {
        queryParams.append(key, value.toString());
      }
    });

    const endpoint = `/users${queryParams.toString() ? `?${queryParams}` : ''}`;
    const response = await this.get(endpoint);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as UserListResponse;
  }

  async updateUser(userId: string, userData: Partial<User>) {
    const response = await this.put(`/users/${userId}`, userData);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async patchUser(userId: string, userData: Partial<User>) {
    const response = await this.patch(`/users/${userId}`, userData);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async deleteUser(userId: string) {
    const response = await this.delete(`/users/${userId}`);
    await this.expectSuccessResponse(response, 204);
  }

  async activateUser(userId: string) {
    const response = await this.post(`/users/${userId}/activate`);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async deactivateUser(userId: string) {
    const response = await this.post(`/users/${userId}/deactivate`);
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response) as User;
  }

  async validateUserExists(email: string) {
    const users = await this.getAllUsers({ search: email });
    return users.users.some(user => user.email === email);
  }

  async expectUserNotFound(userId: string) {
    const response = await this.get(`/users/${userId}`);
    await this.expectErrorResponse(response, 404);
  }
}

// api/AuthApiClient.ts
import { APIRequestContext } from '@playwright/test';
import { BaseApiClient } from './BaseApiClient';

export interface LoginCredentials {
  email: string;
  password: string;
  rememberMe?: boolean;
}

export interface AuthResponse {
  token: string;
  refreshToken: string;
  user: {
    id: string;
    name: string;
    email: string;
    role: string;
  };
  expiresIn: number;
}

export interface RegisterData {
  name: string;
  email: string;
  password: string;
  confirmPassword: string;
}

export class AuthApiClient extends BaseApiClient {
  private authToken: string | null = null;

  constructor(apiContext: APIRequestContext) {
    super(apiContext, '/api/auth');
  }

  async login(credentials: LoginCredentials) {
    const response = await this.post('/login', credentials);
    await this.expectSuccessResponse(response);
    
    const authData = await this.getResponseBody(response) as AuthResponse;
    this.authToken = authData.token;
    
    return authData;
  }

  async register(userData: RegisterData) {
    const response = await this.post('/register', userData);
    await this.expectSuccessResponse(response, 201);
    return await this.getResponseBody(response) as AuthResponse;
  }

  async logout() {
    const response = await this.post('/logout', {}, {
      headers: this.getAuthHeaders()
    });
    await this.expectSuccessResponse(response);
    this.authToken = null;
  }

  async refreshToken(refreshToken: string) {
    const response = await this.post('/refresh', { refreshToken });
    await this.expectSuccessResponse(response);
    
    const authData = await this.getResponseBody(response) as AuthResponse;
    this.authToken = authData.token;
    
    return authData;
  }

  async getCurrentUser() {
    const response = await this.get('/me', {
      headers: this.getAuthHeaders()
    });
    await this.expectSuccessResponse(response);
    return await this.getResponseBody(response);
  }

  async changePassword(currentPassword: string, newPassword: string) {
    const response = await this.post('/change-password', {
      currentPassword,
      newPassword
    }, {
      headers: this.getAuthHeaders()
    });
    await this.expectSuccessResponse(response);
  }

  async requestPasswordReset(email: string) {
    const response = await this.post('/password-reset-request', { email });
    await this.expectSuccessResponse(response);
  }

  async resetPassword(token: string, newPassword: string) {
    const response = await this.post('/password-reset', {
      token,
      newPassword
    });
    await this.expectSuccessResponse(response);
  }

  async verifyEmail(token: string) {
    const response = await this.post('/verify-email', { token });
    await this.expectSuccessResponse(response);
  }

  getAuthToken() {
    return this.authToken;
  }

  private getAuthHeaders() {
    return this.authToken ? { 'Authorization': `Bearer ${this.authToken}` } : {};
  }

  async expectUnauthorized(endpoint: string) {
    const response = await this.get(endpoint);
    await this.expectErrorResponse(response, 401);
  }

  async expectForbidden(endpoint: string) {
    const response = await this.get(endpoint, {
      headers: this.getAuthHeaders()
    });
    await this.expectErrorResponse(response, 403);
  }
}
